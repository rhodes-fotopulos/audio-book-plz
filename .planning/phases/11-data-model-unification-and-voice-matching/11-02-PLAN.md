```python
class VoiceProfile(BaseModel):
    model_config = ConfigDict(strict=True)
    # Coarse traits (LibriTTS-P aligned)
    pitch: str       # "high", "medium", "low", "unknown"
    pace: str        # "fast", "moderate", "slow", "unknown"
    tone: str        # e.g. "warm", "gruff", "silky", "unknown"
    accent: str      # e.g. "British", "unknown"
    # Rich descriptors
    pace_style: str  # e.g. "measured", "rapid", "languid", "unknown"
    tone_style: str  # e.g. "gravelly", "melodic", "breathy", "unknown"
    energy: str      # e.g. "restrained", "animated", "unknown"
    typical_emotion: str  # e.g. "sardonic", "cheerful", "unknown"
    description: str # one-sentence voice summary

class CharacterProfile(BaseModel):
    model_config = ConfigDict(strict=True)
    name: str
    aliases: list[str] = []
    gender: str
    age_range: str
    voice_profile: VoiceProfile  # NEW - replaces voice_qualities + voice_baseline
    personality_traits: list[str] = []
    description: str
    is_named: bool
```
